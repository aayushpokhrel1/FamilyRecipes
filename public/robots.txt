# Public pages are crawlable. Everything below Disallow is either behind a sign-in wall or a
# per-user view, so indexing it would only ever produce a search result that 404s or bounces
# a visitor to /signin.

# The marketing template on the bare domain. It is an asset, so it is fetchable by path, but
# it is not a page: it still carries the rack placeholder.
Disallow: /site.html

# The prerendered body fragments the Worker injects into the four public documents. They are
# refused as public paths by the Worker, which is what actually holds; this is belt and
# braces. A fragment has no head and no canonical, so an indexed one would be a context-free
# duplicate of the real page.
Disallow: /prerender/

# The two health routes. Neither is a page: one answers a monitor and the other names the
# commit this Worker was built from. A crawl of /health/extract would send a real request
# upstream on every hit, and an indexed /health/version would publish the deployed commit of a
# public repository where we cannot take it back. The Worker also sends X-Robots-Tag on both,
# which is what actually holds; this is the half that stops the fetch.
Disallow: /health/
